import spacy
nlp = spacy.load("en_core_web_sm")
corpus = """
Barack Obama Visited India in 2015 and met Prime Minister Narendra Modi in New Delhi. He also delivered speech at Siri Fort Auditorium and discussed relations between United States and India. The meeting highlighted Indo-US strategic partnerships in defense, trade and technology."""
if len(corpus.split()) > 70:
  print("Corpus exceeds 70 words.Truncating to first 70 words.")
  corpus = ' '.join(corpus.split()[:70])
doc = nlp(corpus)
if doc.ents:
  print("Named entities found:")
  for ent in doc.ents:
    print(f"{ent.text} --> {ent.label_}")
else:
  print("No named entities found in the corpus")
