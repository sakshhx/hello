(4)# RELATIONAL OPERATORS

a <- c(10, 20, 30, 40)
b <- c(25, 2, 30, 3)

cat(a, "less than", b, (a < b), "\n")
cat(a, "greater than", b, (a > b), "\n")
cat(a, "less than or equal to", b, (a <= b), "\n")
cat(a, "greater than or equal to", b, (a >= b), "\n")
cat(a, "equal to", b, (a == b), "\n")
cat(a, "not equal to", b, (a != b), "\n")

(5)# LOGICAL OPERATORS

a <- c(0, 20, 30, 56)
b <- c(2, 2, 30, 0)

# Logical NOT
cat(a, "logical NOT", (!a), "\n")

# Element-wise logical AND
cat(a, "element-wise logical AND", b, (a & b), "\n")

# Logical AND (NOT allowed for vectors — so show first element only)
cat("logical AND (first elements only)", a[1], b[1], (a[1] && b[1]), "\n")

# Element-wise logical OR
cat(a, "element-wise logical OR", b, (a | b), "\n")

# Logical OR (first elements only)
cat("logical OR (first elements only)", a[1], b[1], (a[1] || b[1]), "\n")
