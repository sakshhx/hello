import numpy as np
import matplotlib.pyplot as plt

# Sample data: Height (in inches) and Weight (in pounds → converted to kg)
X = np.array([58, 60, 62, 64, 66, 68, 70, 72, 74, 76], dtype=float)
y_pounds = np.array([115, 117, 120, 123, 126, 129, 132, 135, 139, 142], dtype=float)

# Convert weight to kg
y = y_pounds * 0.453592   # 1 lb = 0.453592 kg

# Normalize heights for better gradient descent
X_norm = (X - X.mean()) / X.std()
